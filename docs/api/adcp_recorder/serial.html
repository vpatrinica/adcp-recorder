176    def reconnect(self, max_retries: int = 5, backoff_base: float = 2.0) -> bool:
177        """Attempt to reconnect with exponential backoff.
178
179        Args:
180            max_retries: Maximum number of retry attempts
181            backoff_base: Base for exponential backoff calculation
182
183        Returns:
184            True if reconnection successful, False if all retries exhausted
185
186        Example:
187            >>> manager = SerialConnectionManager('/dev/ttyUSB0')
188            >>> if not manager.is_connected():
189            ...     manager.reconnect(max_retries=3)

191        """
192        self.disconnect()
193
194        for attempt in range(max_retries):
195            if self.connect():

202        return False

105    def _read_loop(self) -> None:
106        """Main read loop (runs in thread)."""
107        logger.info("Producer read loop starting")
108
109        while self._running:
110            # Ensure connection
111            if not self._connection_manager.is_connected():
112                logger.warning("Not connected, attempting reconnection...")
113                if not self._connection_manager.reconnect(max_retries=3):
114                    logger.error("Reconnection failed, waiting before retry")
115                    time.sleep(5.0)
116                    continue
117
118            # Read data
119            line_bytes = self._connection_manager.read_line(timeout=1.0)
120
121            if line_bytes is None:
122                # Timeout or error - continue
123                continue
124
125            if not line_bytes:
126                # Empty line - continue
127                continue
128
129            # Check for binary data
130            if is_binary_data(line_bytes):
131                logger.warning(f"Binary data detected: {line_bytes[:50]!r}")
132                # Enter blob mode and stream binary chunks to the consumer
133                if not self._blob_mode:
134                    self._blob_mode = True
135                    chunk = BinaryChunk(data=line_bytes, start=True)
136                else:
137                    chunk = BinaryChunk(data=line_bytes)
138
139                self._push_to_queue(chunk)
140                self._update_heartbeat()
141                continue
142
143            # Decode and buffer
144            try:
145                line_str = line_bytes.decode("ascii").strip()
146            except UnicodeDecodeError:
147                logger.warning(f"Failed to decode ASCII: {line_bytes[:50]!r}")
148                # Treat as binary (enter blob mode)
149                if not self._blob_mode:
150                    self._blob_mode = True
151                    chunk = BinaryChunk(data=line_bytes, start=True)
152                else:
153                    chunk = BinaryChunk(data=line_bytes)
154
155                self._push_to_queue(chunk)
156                self._update_heartbeat()
157                continue
158
159            # Push complete line
160            if line_str:
161                # If we were in blob mode but now have a printable line, mark blob end
162                if self._blob_mode:
163                    # push explicit end marker then the recovered line
164                    end_chunk = BinaryChunk(data=b"", end=True)
165                    self._push_to_queue(end_chunk)
166                    self._blob_mode = False
167
168                self._push_to_queue(line_str.encode("ascii"))
169                self._update_heartbeat()
170
171        logger.info("Producer read loop exiting")
172
173    def _push_to_queue(self, data: bytes | BinaryChunk) -> None:
174        """Push data to queue, dropping oldest if full.
175
176        Args:
177            data: Line data to push (as bytes or BinaryChunk)
178
179        """
180        try:
181            # Non-blocking put
182            self._queue.put_nowait(data)
183        except Full:
184            # Queue is full - drop oldest and retry
185            logger.warning("Queue full, dropping oldest item")
186            try:
187                self._queue.get_nowait()  # Remove oldest
188                self._queue.put_nowait(data)  # Add new
189            except Exception as e:
190                logger.error(f"Failed to push to queue: {e}")

204    def _consume_loop(self) -> None:
205        """Main consume loop (runs in thread)."""
206        logger.info("Consumer loop starting")
207        conn = self._db_manager.get_connection()
208
209        try:
210            while True:
211                try:
212                    # Pull from queue - use short timeout if stopping to drain quickly
213                    limit = 1.0 if self._running else 0.1
214                    item = self._queue.get(timeout=limit)
215                except Empty:
216                    # Queue empty - if we are stopping, we are done
217                    if not self._running:
218                        logger.debug("Queue empty and stopping, exiting loop")
219                        break
220
221                    # Periodic flush of stale buffers
222                    if self._file_writer:
223                        try:
224                            self._file_writer.flush_stale(max_age_seconds=300)
225                        except Exception as e:
226                            logger.error(f"Failed to flush stale Parquet records: {e}")
227                    continue
228                except Exception as e:
229                    logger.error(f"Unexpected error getting from queue: {e}", exc_info=True)
230                    continue
231                # Binary chunk streaming handling
232                try:
233                    if isinstance(item, BinaryChunk):
234                        # Start of blob
235                        if item.start:
236                            # record parse error/marker once at blob start
237                            insert_parse_error(
238                                conn,
239                                "<BINARY_BLOB>",
240                                error_type="BINARY_DATA",
241                                error_message="Binary blob captured",
242                            )
243                            insert_raw_line(
244                                conn,
245                                "<BINARY_BLOB>",
246                                parse_status="FAIL",
247                                error_message="Binary blob captured",
248                            )
249                            # start file
250                            self._binary_writer.start_blob(item.data)
251                            if self._file_writer:
252                                self._file_writer.write_invalid_record(
253                                    "BINARY", item.data.decode("ascii", errors="replace")
254                                )
255                                # Add to parquet
256                                self._file_writer.write_record(
257                                    "errors",
258                                    {
259                                        "raw_content": item.data.decode("ascii", errors="replace"),
260                                        "error_type": "BINARY_DATA",
261                                        "error_message": "Binary blob captured",
262                                        "attempted_prefix": "BINARY",
263                                    },
264                                )
265                                self._file_writer.write_record(
266                                    "raw_lines",
267                                    {
268                                        "content": item.data.decode("ascii", errors="replace"),
269                                        "parse_status": "FAIL",
270                                        "record_type": "BINARY",
271                                        "error_message": "Binary blob captured",
272                                    },
273                                )
274                        elif item.end:
275                            path = self._binary_writer.finish_blob()
276                            logger.info(f"Binary blob saved to {path}")
277                        else:
278                            # middle chunk
279                            self._binary_writer.append_chunk(item.data)
280
281                        self._update_heartbeat()
282                        continue
283
284                    # Otherwise it's a normal line (bytes)
285                    line_bytes = item
286
287                    # Process line
288                    try:
289                        self._process_line(conn, line_bytes)
290                    except Exception as e:
291                        logger.error(f"Error processing line: {e}", exc_info=True)
292                        # Try to keep going
293                except Exception as e:
294                    logger.error(f"Consumer loop processing error: {e}", exc_info=True)
295
296                try:
297                    self._update_heartbeat()
298                except Exception as e:
299                    logger.error(f"Heartbeat update failed: {e}")
300        finally:
301            with contextlib.suppress(Exception):
302                self._binary_writer.finish_blob()
303            self._db_manager.close()
304            logger.info("Consumer loop exiting")

376    def _process_line(self, conn: duckdb.DuckDBPyConnection, line_bytes: bytes) -> None:
377        """Process a single line from the queue.
378
379        Args:
380            conn: Database connection
381            line_bytes: Line data as bytes
382
383        """
384        # 1. Check for binary data
385        if is_binary_data(line_bytes):
386            self._record_error(
387                conn, line_bytes, "BINARY_DATA", "Binary data detected", prefix="BINARY"
388            )
389            return
390
391        # 2. Decode to string
392        try:
393            sentence = line_bytes.decode("ascii").strip()
394        except UnicodeDecodeError as e:
395            self._record_error(conn, line_bytes, "DECODE_ERROR", str(e))
396            return
397
398        if not sentence:
399            return
400
401        # 3. Basic NMEA validation and Routing
402        prefix = "UNKNOWN"
403        try:
404            # Extract prefix carefully
405            try:
406                prefix = extract_prefix(sentence)
407            except ValueError:
408                # Still try to route/handle if prefix extraction fails (e.g. malformed start)
409                pass
410
411            # Checksum validation if present
412            if "*" in sentence:
413                try:
414                    if not validate_checksum(sentence):
415                        data_part, provided = sentence.rsplit("*", 1)
416                        computed = compute_checksum(sentence)
417                        self._record_error(
418                            conn,
419                            sentence,
420                            "CHECKSUM_ERROR",
421                            f"Checksum mismatch: expected {computed}, got {provided}",
422                            prefix=prefix,
423                            checksum_expected=computed,
424                            checksum_actual=provided,
425                        )
426                        return
427                except ValueError as e:
428                    self._record_error(conn, sentence, "CHECKSUM_ERROR", str(e), prefix=prefix)
429                    return
430
431            # Route to parser
432            parsed = self._router.route(sentence)
433
434            if parsed is None:
435                # Unknown message type - log and store as PENDING
436                logger.debug(f"Unknown message type: {prefix}")
437                insert_raw_line(
438                    conn,
439                    sentence,
440                    parse_status="PENDING",
441                    record_type=prefix,
442                    checksum_valid=True if "*" in sentence else None,
443                    error_message=f"No parser for {prefix}",
444                )
445                if self._file_writer:
446                    with contextlib.suppress(Exception):
447                        self._file_writer.write(prefix, sentence)
448                    with contextlib.suppress(Exception):
449                        self._file_writer.write_record(
450                            "raw_lines",
451                            {
452                                "content": sentence,
453                                "parse_status": "PENDING",
454                                "record_type": prefix,
455                                "error_message": f"No parser for {prefix}",
456                                "checksum_valid": True if "*" in sentence else None,
457                            },
458                        )
459                return
460
461            # 4. Successfully parsed - insert to database
462            self._store_parsed_message(conn, sentence, prefix, parsed)
463
464            # Also insert to raw_lines
465            insert_raw_line(
466                conn,
467                sentence,
468                parse_status="OK",
469                record_type=prefix,
470                checksum_valid=True if "*" in sentence else None,
471            )
472
473            if self._file_writer:
474                with contextlib.suppress(Exception):
475                    self._file_writer.write(prefix, sentence)
476                with contextlib.suppress(Exception):
477                    self._file_writer.write_record(
478                        "raw_lines",
479                        {
480                            "content": sentence,
481                            "parse_status": "OK",
482                            "record_type": prefix,
483                            "checksum_valid": True if "*" in sentence else None,
484                        },
485                    )

487        except ValueError as e:
488            # Parse failed (ValidationError from from_nmea)
489            self._record_error(conn, sentence, "PARSE_ERROR", str(e), prefix=prefix)
