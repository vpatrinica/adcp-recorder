def compute_checksum(sentence: str) -> str: 
 8def compute_checksum(sentence: str) -> str:
 9    """Compute NMEA checksum (XOR of characters between $ and *).
10
11    Args:
12        sentence: NMEA sentence, with or without checksum
13
14    Returns:
15        Two-character uppercase hex checksum (e.g., "2E")
16
17    Example:
18        >>> compute_checksum("$PNORI,4,Signature1000900001,4,20,0.20,1.00,0*2E")
19        '2E'
20
21    """
22    # Remove $ prefix and everything after * (including checksum)
23    if sentence.startswith("$"):
24        sentence = sentence[1:]
25
26    if "*" in sentence:
27        sentence = sentence.split("*", 1)[0]
28
29    # XOR all characters
30    checksum = 0
31    for char in sentence:
32        checksum ^= ord(char)
33
34    return f"{checksum:02X}"

 90    """Split NMEA sentence into fields and checksum.
 91
 92    Args:
 93        sentence: Complete NMEA sentence
 94
 95    Returns:
 96        Tuple of (fields_list, checksum_string)
 97        Fields list includes the $PREFIX as first element
 98        Checksum is None if not present
 99
100    Example:
101        >>> split_sentence("$PNORI,4,Test,4,20,0.20,1.00,0*2E")
102        (['$PNORI', '4', 'Test', '4', '20', '0.20', '1.00', '0'], '2E')
103
104    """
105    sentence = sentence.strip()
106
107    # Extract checksum if present
108    checksum = None
109    if "*" in sentence:
110        data_part, checksum = sentence.rsplit("*", 1)
111        checksum = checksum.strip().upper()
112    else:
113        data_part = sentence
114
115    # Split fields and trim whitespace
116    fields = [f.strip() for f in data_part.split(",")]
117
118    return fields, checksum

def is_binary_data(data: bytes, threshold: int = 1024) -> bool: 
121def is_binary_data(data: bytes, threshold: int = 1024) -> bool:
122    """Detect if data contains binary (non-NMEA) content.
123
124    NMEA sentences should be printable ASCII (0x20-0x7E) plus CR/LF.
125    This function detects binary blobs by counting non-printable characters.
126
127    Args:
128        data: Bytes to check
129        threshold: Max bytes to scan before detection (default 1024)
130
131    Returns:
132        True if data appears to be binary, False if it looks like NMEA
133
134    Note:
135        Per spec, if more than MAX_NON_NMEA_CHARS_PER_LINE non-printable
136        characters are found, we consider it binary data.
137
138    """
139    # NMEA valid characters: printable ASCII (0x20-0x7E) + CR (0x0D) + LF (0x0A)
140    scan_length = min(len(data), threshold)
141    non_nmea_count = 0
142
143    # Count non-NMEA characters
144    for byte in data[:scan_length]:
145        # Allow printable ASCII, CR, LF
146        if not (0x20 <= byte <= 0x7E or byte in (0x0D, 0x0A)):
147            non_nmea_count += 1
148
149    # If more than 10% of scanned bytes are non-NMEA, treat as binary
150    max_allowed = scan_length // 10
151
152    return non_nmea_count > max_allowed

 62    Values:
 63        ENU: East-North-Up coordinates
 64        XYZ: Instrument XYZ coordinates
 65        BEAM: Raw beam coordinates
 66    """
 67
 68    ENU = "ENU"
 69    XYZ = "XYZ"
 70    BEAM = "BEAM"
 71
 72    @classmethod
 73    def from_code(cls, code: int | str) -> "CoordinateSystem":
 74        """Create coordinate system from code.
 75
 76        Args:
 77            code: Numeric code (0, 1, 2) or string ("ENU", "XYZ", "BEAM")
 78
 79        Returns:
 80            CoordinateSystem enum value
 81
 82        Raises:
 83            ValueError: If code is not valid
 84
 85        """
 86        if isinstance(code, int):
 87            mapping = {0: cls.ENU, 1: cls.XYZ, 2: cls.BEAM}
 88            if code not in mapping:
 89                raise ValueError(
 90                    f"Invalid coordinate system code: {code}. "
 91                    f"Valid codes: 0 (ENU), 1 (XYZ), 2 (BEAM)"
 92                )
 93            return mapping[code]
 94        # String code
 95        code_upper = str(code).upper()
 96        try:
 97            return cls(code_upper)
 98        except ValueError:
 99            raise ValueError(f"Invalid coordinate system: {code}. Valid values: ENU, XYZ, BEAM")
100
101    def to_numeric_code(self) -> int:
102        """Convert to numeric code for backward compatibility.
103
104        Returns:
105            Numeric code: 0 (ENU), 1 (XYZ), 2 (BEAM)
106
107        """
108        mapping = {self.ENU: 0, self.XYZ: 1, self.BEAM: 2}
109        return mapping[self]
