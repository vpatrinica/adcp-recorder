 27        """Initialize database manager.

 67        """Create all required tables and indexes if they don't exist.
 68
 69        This method is idempotent and safe to call multiple times.
 70        Uses IF NOT EXISTS clauses to avoid errors on existing tables.

 92    def initialize_ducklake(self) -> None:
 93        """Initialize DuckLake views pointing to Parquet files if they exist."""
 94        conn = self.get_connection()
 95
 96        # We look for parquet files in the standard output directory
 97        # This assumes the DB is sibling to 'parquet' folder or we can derive it
 98        # Real implementation should probably get this from config
 99        base_path = Path(self.db_path).parent
100        parquet_path = base_path / "parquet"
101
102        if not parquet_path.exists():
103            return
104
105        logger.info(f"Initializing DuckLake views from {parquet_path}")
106
107        # Register a view for each record type found in parquet folder
108        try:
109            for record_type_dir in parquet_path.iterdir():
110                if record_type_dir.is_dir():
111                    prefix = record_type_dir.name.lower()
112                    view_name = f"view_{prefix}"
113                    parquet_glob = str(record_type_dir / "**" / "*.parquet")
114
115                    conn.execute(
116                        f"CREATE OR REPLACE VIEW {view_name} "
117                        f"AS SELECT * FROM read_parquet('{parquet_glob}')"
118                    )
119                    logger.debug(f"Created DuckLake view: {view_name}")
120        except Exception as e:
121            logger.warning(f"Failed to initialize some DuckLake views: {e}")

566    in_place: bool = False,

568    """Perform full database migration from v0.1.x to v0.2.0 schema.

586    else:
587        if target_path is None:
588            target_path = source_path.parent / f"{source_path.stem}_migrated.duckdb"

592        import shutil
593
594        logger.info(f"Copying {source_path} to {target_path}")
595        shutil.copy2(source_path, target_path)
596
597    # Connect to target database
598    conn = duckdb.connect(str(target_path))
599
600    stats: dict[str, int] = {}
601
602    try:
603        from adcp_recorder.db.schema import TABLE_SCHEMA_SQL, VIEW_SCHEMA_SQL
604
605        # 0. Drop all views first to avoid dependency blocks
606        drop_all_views(conn)
607
608        # 1. Fix existing tables that might have typos or old field names
609        # These MUST happen before views are created to avoid Dependency Errors
610        # and Binder Errors.
611        stats["pnorw_data (field update)"] = migrate_pnorw_fields(conn)
612        stats["pnorw_data (h3 fix)"] = ensure_pnorw_h3(conn)
613        stats["pnorb_data (typo fix)"] = fix_pnorb_typos(conn)

615        # 2. Create new schema tables only

617        create_new_schema(conn, TABLE_SCHEMA_SQL)
