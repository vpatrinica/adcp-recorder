 13class FileWriter:
 14    """Writes NMEA sentences to files and structured records to Parquet.
 15
 16    Maintains separate files for each message type and rotates daily.
 17    Also handles Parquet export for structured ADCP data.
 18    """

 20    def __init__(self, base_path: str):
 21        """Initialize file writer.
 22
 23        Args:
 24            base_path: Base directory to write files to
 25
 26        """
 27        self.base_path = base_path
 28        self._files: dict[str, TextIO] = {}
 29        self._current_date = datetime.now().date()
 30        self._closed = False
 31        self._ensure_base_path()
 32        self.parquet_writer = ParquetWriter(base_path)
 33
 34    def _ensure_base_path(self) -> None:
 35        """Ensure base directory exists."""
 36        os.makedirs(self.base_path, exist_ok=True)
 37
 38    def _get_filename(self, prefix: str) -> str:
 39        """Get filename for a message type and current date."""
 40
 41        # Format: nmea/{prefix}/{prefix}_{YYYYMMDD}.nmea
 42        date_str = self._current_date.strftime("%Y%m%d")
 43
 44        # Ensure directory exists
 45        file_dir = os.path.join(self.base_path, "nmea", prefix)
 46        os.makedirs(file_dir, exist_ok=True)
 47
 48        return os.path.join(file_dir, f"{prefix}_{date_str}.nmea")
 49
 50    def _check_rotation(self) -> None:
 51        """Check if files need to be rotated (new day)."""
 52        now = datetime.now().date()
 53        if now != self._current_date:
 54            logger.info(f"Rotating files from {self._current_date} to {now}")
 55            # Do not call self.close() here as it sets _closed=True
 56            # Instead, just close internal handles
 57            self._close_handles()
 58            self._current_date = now
 59            self._files = {}
 60
 61    def _get_file_handle(self, prefix: str) -> TextIO | None:
 62        """Get or create file handle for message type."""
 63        if self._closed:
 64            return None
 65
 66        self._check_rotation()
 67
 68        if prefix not in self._files:
 69            filename = self._get_filename(prefix)
 70            try:
 71                self._files[prefix] = open(filename, "a", encoding="utf-8", buffering=1)
 72                logger.debug(f"Opened log file: {filename}")
 73            except OSError as e:
 74                logger.error(f"Failed to open log file {filename}: {e}")
 75                return None
 76
 77        return self._files[prefix]
 78
 79    def write(self, prefix: str, data: str) -> None:
 80        """Write data to file for message type.
 81
 82        Args:
 83            prefix: Message type prefix (e.g. 'PNORI') or 'ERRORS'
 84            data: Data string to write (should include newlines if needed)
 85
 86        """
 87        if self._closed or not prefix or not data:
 88            return
 89
 90        try:
 91            f = self._get_file_handle(prefix)
 92            if f:
 93                f.write(data)
 94                if not data.endswith("\n"):
 95                    f.write("\n")
 96                f.flush()
 97        except Exception as e:
 98            logger.error(f"Failed to write to file for {prefix}: {e}")
 99
100    def write_record(self, prefix: str, record: dict[str, Any]) -> None:
101        """Write structured record to Parquet.
102
103        Args:
104            prefix: Message type prefix
105            record: Data dictionary
106
107        """
108        if self._closed:
109            return
110
111        try:
112            self.parquet_writer.write_record(prefix, record)
113        except Exception as e:
114            logger.error(f"Failed to write record for {prefix} to Parquet: {e}")
115
116    def flush_stale(self, max_age_seconds: int = 300) -> None:
117        """Flush stale Parquet buffers.
118
119        Args:
120            max_age_seconds: Threshold for flushing stale buffers.
121        """
122        if self._closed:
123            return  # pragma: no cover
124
125        try:
126            self.parquet_writer.flush_stale(max_age_seconds)
127        except Exception as e:
128            logger.error(f"Failed to flush stale Parquet records: {e}")
129
130    def write_invalid_record(self, prefix: str, data: str) -> None:
131        """Write invalid record to error file.
132
133        Args:
134            prefix: Message type prefix
135            data: Invalid data string
136
137        """
138        if self._closed or not prefix or not data:
139            return
140
141        try:
142            date_str = self._current_date.strftime("%d%m%y")
143            error_dir = os.path.join(self.base_path, "errors", "nmea")
144            os.makedirs(error_dir, exist_ok=True)
145
146            # Consolidate BINARY and unknown prefixes into ERROR_{date}.nmea
147            # Known prefixes (PNORI, PNORS, etc.) use {prefix}_error_{date}.nmea
148            known_prefixes = {
149                "PNORI",
150                "PNORS",
151                "PNORC",
152                "PNORH",
153                "PNORW",
154                "PNORB",
155                "PNORE",
156                "PNORF",
157                "PNORWD",
158                "PNORA",
159            }
160            if prefix == "BINARY" or prefix not in known_prefixes:
161                filename = os.path.join(error_dir, f"ERROR_{date_str}.nmea")
162            else:
163                filename = os.path.join(error_dir, f"{prefix}_error_{date_str}.nmea")
164
165            # Append to file
166            # Note: For error files we open/close immediately to avoid keeping many handles open
167            # This is less efficient but error rate should be low
168            with open(filename, "a", encoding="utf-8", buffering=1) as f:
169                f.write(data)
170                if not data.endswith("\n"):
171                    f.write("\n")
172                f.flush()
173        except Exception as e:
174            logger.error(f"Failed to write invalid record for {prefix}: {e}")
175
176    def write_error(self, message: str) -> None:
177        """Write error message to error log."""
178        timestamp = datetime.now().isoformat()
179        self.write_invalid_record("SYSTEM", f"[{timestamp}] {message}")
180
181    def _close_handles(self) -> None:
182        """Close all open file handles."""
183        for prefix, f in self._files.items():
184            try:
185                f.close()
186            except Exception as e:
187                logger.error(f"Error closing file for {prefix}: {e}")
188
189    def close(self) -> None:
190        """Close all open files and writers."""
191        self._closed = True
192        self._close_handles()
193        self._files.clear()
194
195        try:
196            self.parquet_writer.close()
197        except Exception as e:
198            logger.error(f"Error closing parquet writer: {e}")
