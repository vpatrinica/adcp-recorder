 14class ParquetWriter:
 15    """Writes structured records to Parquet files with daily partitioning.
 16
 17    Uses DuckDB as the engine for efficient Parquet generation.

 20    def __init__(self, base_path: str, buffer_size: int = 100):
 21        """Initialize Parquet writer.
 22
 23        Args:
 24            base_path: Base directory for the "DuckLake" storage
 25            buffer_size: Number of records to buffer before flushing to disk
 26
 27        """
 28        self.base_path = Path(base_path) / "parquet"
 29        self.buffer_size = buffer_size
 30        self._buffers: dict[str, list[dict[str, Any]]] = {}
 31        self._last_flush: dict[str, datetime] = {}
 32        self._conn = duckdb.connect(database=":memory:")
 33        self._ensure_base_path()
 34
 35    def _ensure_base_path(self) -> None:
 36        """Ensure base directory for Parquet files exists."""
 37        os.makedirs(self.base_path, exist_ok=True)
 38
 39    def _get_partition_path(self, prefix: str, record_date: date) -> Path:
 40        """Get the partitioned directory path for a record type and date."""
 41        # Partitioning by record type and then by date
 42        # Format: base/prefix/date=YYYY-MM-DD/
 43        partition_dir = self.base_path / prefix / f"date={record_date.isoformat()}"
 44        os.makedirs(partition_dir, exist_ok=True)
 45        return partition_dir

 47    def write_record(self, prefix: str, record: dict[str, Any]) -> None:
 48        """Buffer a record for writing.
 49
 50        Args:
 51            prefix: Record type prefix (e.g., 'PNORS', 'PNORC')
 52            record: Dictionary of data to store
 53
 54        """
 55        if prefix not in self._buffers:
 56            self._buffers[prefix] = []
 57            self._last_flush[prefix] = datetime.now()
 58
 59        # Add timestamp if not present
 60        if "received_at" not in record:
 61            record["received_at"] = datetime.now()
 62
 63        # Add measurement_id for optimized joins if date and time are present
 64        if "measurement_date" in record and "measurement_time" in record:
 65            try:
 66                date_str = str(record["measurement_date"])
 67                time_str = str(record["measurement_time"])
 68                if len(date_str) == 6 and len(time_str) == 6:
 69                    # NMEA date is MMDDYY, we want YYMMDDHHMMSS for sorting
 70                    yy = date_str[4:6]
 71                    mm = date_str[0:2]
 72                    dd = date_str[2:4]
 73                    record["measurement_id"] = int(f"{yy}{mm}{dd}{time_str}")
 74            except (ValueError, TypeError):
 75                pass
 76
 77        self._buffers[prefix].append(record)
 78
 79        if len(self._buffers[prefix]) >= self.buffer_size:
 80            self.flush(prefix)
 81        else:
 82            # Check for stale buffers occasionally even if not full
 83            # We don't want to check on every single record for high-freq ones,
 84            # but for low-freq it's fine.
 85            self.flush_stale(max_age_seconds=300)  # 5 minutes default
 86
 87    def flush_stale(self, max_age_seconds: int = 300) -> None:
 88        """Flush buffers that haven't been flushed in a while.
 89
 90        Args:
 91            max_age_seconds: Threshold for flushing stale buffers.
 92        """
 93        now = datetime.now()
 94        for prefix in list(self._buffers.keys()):
 95            if not self._buffers[prefix]:
 96                continue
 97
 98            last_flush = self._last_flush.get(prefix)
 99            if last_flush is None or (now - last_flush).total_seconds() > max_age_seconds:
100                logger.info(f"Time-based flush for {prefix} ({len(self._buffers[prefix])} records)")
101                self.flush(prefix)
102
103    def flush(self, prefix: str | None = None) -> None:
104        """Flush buffered records to Parquet files.
105
106        Args:
107            prefix: If specified, only flush that prefix. Otherwise flush all.
108
109        """
110        prefixes = [prefix] if prefix else list(self._buffers.keys())
111
112        for p in prefixes:
113            buffer = self._buffers.get(p)
114            if not buffer:
115                continue
116
117            try:
118                # Group by date for partitioning
119                records_by_date: dict[date, list[dict[str, Any]]] = {}
120                for rec in buffer:
121                    ts = rec.get("received_at")
122                    date_val = ts.date() if isinstance(ts, datetime) else datetime.now().date()
123                    if date_val not in records_by_date:
124                        records_by_date[date_val] = []
125                    records_by_date[date_val].append(rec)
126
127                for date_val, records in records_by_date.items():
128                    self._write_to_parquet(p, date_val, records)
129
130                self._buffers[p] = []
131                self._last_flush[p] = datetime.now()
132            except Exception as e:
133                logger.error(f"Failed to flush Parquet records for {p}: {e}")

135    def _write_to_parquet(
136        self, prefix: str, record_date: date, records: list[dict[str, Any]]
137    ) -> None:
138        """Actually write a batch of records to a Parquet file.
139
140        Ensures maximum 1 file per day per prefix by appending to existing file if it exists.
141        Also compacts legacy files ({prefix}_*.parquet) into the single daily file.
142        Uses atomic write signaling: writes to a temporary .writing file first,
143        then renames to the final .parquet file. This ensures readers never see
144        incomplete files.
145        """
146        partition_dir = self._get_partition_path(prefix, record_date)
147
148        # Filename: {prefix}.parquet (one per daily partition)
149        final_filename = f"{prefix}.parquet"
150        temp_filename = f"{prefix}.parquet.writing"
151
152        final_path = partition_dir / final_filename
153        temp_path = partition_dir / temp_filename
154
155        # Detect legacy files: {prefix}_*.parquet (but not the final one)
156        legacy_files = [
157            f
158            for f in partition_dir.glob(f"{prefix}_*.parquet")
159            if f.name != final_filename and not f.name.endswith(".writing")
160        ]
161
162        # Ensure all records have record_type for consistency
163        for r in records:
164            if "record_type" not in r:
165                r["record_type"] = prefix
166
167        try:
168            # Use polars to write Parquet directly - more efficient and no pandas dependency
169            import polars as pl
170
171            # Create dataframe for new records
172            new_df = pl.from_dicts(records, infer_schema_length=10000)
173
174            # Collect data from existing and legacy files
175            dataframes = []
176            if final_path.exists():
177                try:
178                    dataframes.append(pl.read_parquet(str(final_path)))
179                except Exception as e:
180                    logger.warning(f"Could not read existing Parquet {final_path}: {e}")
181
182            for legacy_path in legacy_files:
183                try:
184                    dataframes.append(pl.read_parquet(str(legacy_path)))
185                except Exception as e:
186                    logger.warning(f"Could not read legacy Parquet {legacy_path}: {e}")
187
188            dataframes.append(new_df)
189
190            # Align types to handle schema conflicts (e.g., Int64 vs Float64)
191            if dataframes:
192                # Get common columns
193                all_cols = set(new_df.columns)
194                for df_existing in dataframes[:-1]:  # new_df is at the end
195                    all_cols.update(df_existing.columns)
196
197                # For all columns in new_df, align with existing dataframes if types differ
198                for col in new_df.columns:
199                    for i, df_existing in enumerate(dataframes[:-1]):
200                        if col in df_existing.columns:
201                            existing_dtype = df_existing[col].dtype
202                            new_dtype = new_df[col].dtype
203                            if new_dtype != existing_dtype:
204                                try:
205                                    # Handle Null types on either side:
206                                    # Polars cannot cast to/from Null, so rebuild
207                                    # the Null-typed column with the real type.
208                                    if "Null" in str(new_dtype):
209                                        # New data is Null → adopt existing type
210                                        new_df = new_df.with_columns(
211                                            pl.lit(None, dtype=existing_dtype).alias(col)
212                                        )
213                                    elif "Null" in str(existing_dtype):
214                                        # Existing file is Null → adopt new type
215                                        dataframes[i] = df_existing.with_columns(
216                                            pl.lit(None, dtype=new_dtype).alias(col)
217                                        )
218                                    # If target is float-ish, promote to float
219                                    elif "Float" in str(existing_dtype):
220                                        new_df = new_df.with_columns(
221                                            pl.col(col).cast(pl.Float64)
222                                        )  # pragma: no cover
223                                    else:
224                                        new_df = new_df.with_columns(
225                                            pl.col(col).cast(existing_dtype)
226                                        )
227                                except Exception as e:
228                                    logger.warning(
229                                        f"Failed to align column {col} type from "
230                                        f"{new_df[col].dtype} to {existing_dtype}: {e}"
231                                    )
232                                break
233                            break
234
235                # Update the reference in the dataframes list to use the aligned version
236                dataframes[-1] = new_df
237
238            # Use diagonal union to handle schema evolutions (new columns)
239            df = pl.concat(dataframes, how="diagonal")
240
241            # Backfill measurement_id for legacy records if missing
242            if "measurement_id" in df.columns:
243                # Fill nulls in measurement_id if date and time are present
244                # This is a bit complex in polars expressions, so we do it if possible
245                if "measurement_date" in df.columns and "measurement_time" in df.columns:
246                    # Only try if we have the necessary strings
247                    df = df.with_columns(
248                        pl.when(pl.col("measurement_id").is_null())
249                        .then(
250                            # YYMMDDHHMMSS
251                            (
252                                pl.col("measurement_date").str.slice(4, 2)
253                                + pl.col("measurement_date").str.slice(0, 2)
254                                + pl.col("measurement_date").str.slice(2, 2)
255                                + pl.col("measurement_time")
256                            ).cast(pl.UInt64)
257                        )
258                        .otherwise(pl.col("measurement_id"))
259                        .alias("measurement_id")
260                    )
261
262            # Write to temporary file first
263            df.write_parquet(str(temp_path))
264
265            # Atomic replace to final path (atomic on POSIX systems, replaces on Windows if exists)
266            os.replace(temp_path, final_path)
267
268            # Clean up legacy files after successful write
269            for legacy_path in legacy_files:
270                try:
271                    legacy_path.unlink()
272                except Exception as e:
273                    logger.warning(f"Could not delete legacy file {legacy_path}: {e}")
274
275            logger.debug(f"Wrote {len(records)} records (total {len(df)}) to {final_path}")
276        except Exception as e:
277            logger.error(f"Polars Parquet write error: {prefix}: {e}")
278            # Clean up temp file if it exists
279            if temp_path.exists():
280                try:
281                    temp_path.unlink()
282                except OSError:
283                    pass
284            raise
